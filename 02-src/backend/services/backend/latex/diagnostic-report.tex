% DiagnosticPro report preamble — locked to /whiteglove-pdf standards.
% Numeric spacing values are copied from the skill template
% (~/.claude/skills/whiteglove-pdf/templates/default.tex). Only the footer
% branding differs (product URL, not partners.intentsolutions.io).
% Loaded via pandoc --include-in-header.

% --- Tables -------------------------------------------------------------
% booktabs: top + header-separator + bottom only. No full grid, no row
% striping, no vertical rules.
\usepackage{booktabs}
\usepackage{longtable}
\usepackage{array}
\usepackage{etoolbox}

% Vertical breathing room before/after EVERY longtable.
\setlength{\LTpre}{2.6em}
\setlength{\LTpost}{2.6em}

% Row breathing room (default 1.0 is cramped at 11pt).
\renewcommand{\arraystretch}{1.35}

% Booktabs rule weights — heavy top/bottom, light header separator.
\setlength{\heavyrulewidth}{0.9pt}
\setlength{\lightrulewidth}{0.4pt}
\setlength{\cmidrulewidth}{0.4pt}
\setlength{\aboverulesep}{0.45ex}
\setlength{\belowrulesep}{0.55ex}

% --- Captions -----------------------------------------------------------
% Bold "Table N." + period separator, left-aligned (whiteglove standard).
% We rely on pandoc `: caption` lines in the source markdown for a single
% caption per table. Do NOT auto-inject a second "Table N." label above
% longtable (that doubled captions on page 1).
\usepackage{caption}
\captionsetup{
  font={normalsize},
  labelfont={bf},
  labelsep=period,
  justification=raggedright,
  singlelinecheck=false,
  skip=0.6em
}

% --- Headings -----------------------------------------------------------
% Generous breathing room above section headings so they don't crowd a
% preceding table bottom rule.
\usepackage{titlesec}
\titlespacing*{\section}{0pt}{3.4ex plus 1ex minus .2ex}{1.6ex plus .2ex}
\titlespacing*{\subsection}{0pt}{2.8ex plus 1ex minus .2ex}{1.2ex plus .2ex}
\titlespacing*{\subsubsection}{0pt}{2.0ex plus 1ex minus .2ex}{1.0ex plus .2ex}

% Prevent orphaned headings at the very bottom of a page.
\makeatletter
\@afterindentfalse
\@beginparpenalty=-\@lowpenalty
\@endparpenalty=-\@lowpenalty
\@itempenalty=-\@lowpenalty
\makeatother

% --- Widow / orphan control --------------------------------------------
\widowpenalty=10000
\clubpenalty=10000
\displaywidowpenalty=10000
\predisplaypenalty=10000

% --- Body density -------------------------------------------------------
% Slight paragraph breathing without turning the report into double-space.
\setlength{\parskip}{0.35em}
\setlength{\parindent}{0pt}

% --- Page footer (DiagnosticPro product branding) -----------------------
% Header intentionally blank. Footer: brand | domain | page-of-pages.
\usepackage{lastpage}
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0.4pt}
\fancyfoot[L]{\footnotesize DiagnosticPro \,\textbullet\, Intent Solutions}
\fancyfoot[C]{\footnotesize diagnosticpro.io}
\fancyfoot[R]{\footnotesize Page \thepage\ of \pageref*{LastPage}}

\fancypagestyle{plain}{%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0.4pt}%
  \fancyfoot[L]{\footnotesize DiagnosticPro \,\textbullet\, Intent Solutions}%
  \fancyfoot[C]{\footnotesize diagnosticpro.io}%
  \fancyfoot[R]{\footnotesize Page \thepage\ of \pageref*{LastPage}}%
}

% --- Unicode glyph fallbacks -------------------------------------------
% Latin Modern lacks some BMP symbols; map common ones to math equivalents.
% Emoji are stripped by scripts/whiteglove-preprocess.py before pandoc.
\usepackage{newunicodechar}
\usepackage{amssymb}
\newunicodechar{≥}{\ensuremath{\geq}}
\newunicodechar{≤}{\ensuremath{\leq}}
\newunicodechar{≠}{\ensuremath{\neq}}
\newunicodechar{≈}{\ensuremath{\approx}}
\newunicodechar{✓}{\ensuremath{\checkmark}}
\newunicodechar{✗}{\ensuremath{\times}}
\newunicodechar{→}{\ensuremath{\rightarrow}}
\newunicodechar{←}{\ensuremath{\leftarrow}}
\newunicodechar{↔}{\ensuremath{\leftrightarrow}}
\newunicodechar{·}{\ensuremath{\cdot}}
\newunicodechar{×}{\ensuremath{\times}}
